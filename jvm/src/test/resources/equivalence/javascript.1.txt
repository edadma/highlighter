export class Greeter extends Base {
  #count = 0;
  static default = new Greeter("world");

  constructor(name = "anon") {
    super();
    this.name = name;
  }

  greet(times) {
    for (let i = 0; i < times; i++) {
      this.#count += 1;
      console.log(`Hello, ${this.name}! (${i + 1}/${times})`);
    }
    return this.#count >= MAX ? null : undefined;
  }
}

async function load(file) {
  try {
    const text = await readFile(path.join(__dirname, file), "utf8");
    return text.split(/\r?\n/g).filter(l => !/^\s*#/.test(l));
  } catch (err) {
    throw new Error(`cannot read ${file}: ${err.message}`);
  }
}
